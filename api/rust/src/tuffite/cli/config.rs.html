23pub(super) fn resolve_manifest(requested: Option<&Path>) -> Result<PathBuf, String> {
24    discover_manifest(requested)?.ok_or_else(|| {
25        "no tuffite.json found in the current directory or its parents; pass --manifest <PATH>"
26            .to_owned()
27    })

30pub(super) fn discover_manifest(requested: Option<&Path>) -> Result<Option<PathBuf>, String> {
31    if let Some(path) = requested {
32        return super::filesystem::canonical_file(path).map(Some);
33    }
34    let current =
35        std::env::current_dir().map_err(super::filesystem::io_error("read current directory"))?;
36    find_manifest(&current)
37}

39fn find_manifest(start: &Path) -> Result<Option<PathBuf>, String> {
40    for directory in start.ancestors() {
41        let path = directory.join("tuffite.json");
42        match fs::symlink_metadata(&path) {
43            Ok(_) => return super::filesystem::canonical_file(&path).map(Some),
44            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
45            Err(error) => return Err(format!("inspect {}: {error}", path.display())),
46        }
47    }
48    Ok(None)
49}

51pub fn check(path: &Path) -> Result<Value, String> {
52    let bytes = fs::read(path).map_err(|error| format!("read {}: {error}", path.display()))?;
53    let root: Value = serde_json::from_slice(&bytes)
54        .map_err(|error| format!("parse {}: {error}", path.display()))?;
55    let root = object(&root, "$")?;
56    known_fields(
57        root,
58        "$",
59        &["$schema", "productName", "app", "build", "bundle"],
60    )?;
61    if let Some(name) = root.get("productName") {
62        validate_product_name(nonempty_string(Some(name), "productName")?)?;
63    }
64    let build = required(root, "build", "build")?;
65    check_build(build)?;
66    check_app(
67        required(root, "app", "app")?,
68        build["shellApi"]["namespace"].as_str().unwrap(),
69    )?;
70    if let Some(bundle) = root.get("bundle") {
71        check_bundle(bundle)?;
72    }
73    if root["app"]["security"]["capabilities"]
74        .as_array()
75        .is_some_and(|rules| {
76            rules.iter().any(|rule| {
77                rule.get("shellApi")
78                    .and_then(Value::as_array)
79                    .is_some_and(|entries| !entries.is_empty())
80            })
81        })
82    {
83        crate::build::security_command_ids(path)?;
84    }
85    Ok(Value::Object(root.clone()))
86}

88pub(super) fn validate_product_name(value: &str) -> Result<(), String> {
89    if value.len() > 128 {
90        return Err("productName must not exceed 128 UTF-8 bytes".to_owned());
91    }
92    validate_output_name(value).map_err(|error| error.replace("output name", "productName"))
93}

95pub(super) fn validate_output_name(value: &str) -> Result<(), String> {
96    let stem = value
97        .split('.')
98        .next()
99        .unwrap_or(value)
100        .to_ascii_uppercase();
101    if value.trim().is_empty()
102        || value.len() > 240
103        || value.ends_with(['.', ' '])
104        || value.chars().any(|c| {
105            c.is_control() || matches!(c, '/' | '\\' | ':' | '*' | '?' | '"' | '<' | '>' | '|')
106        })
107        || matches!(stem.as_str(), "CON" | "PRN" | "AUX" | "NUL")
108        || (stem.len() == 4
109            && (stem.starts_with("COM") || stem.starts_with("LPT"))
110            && stem.as_bytes()[3].is_ascii_digit())
111    {
112        return Err("output name must be a portable filename of 1..=240 bytes without reserved characters or names".to_owned());
113    }
114    Ok(())
115}

117fn check_app(value: &Value, namespace: &str) -> Result<(), String> {
118    let app = object(value, "app")?;
119    known_fields(
120        app,
121        "app",
122        &[
123            "identifier",
124            "windows",
125            "assets",
126            "network",
127            "utilities",
128            "security",
129        ],
130    )?;
131    if let Some(utilities) = app.get("utilities") {
132        for (name, definition) in object(utilities, "app.utilities")? {
133            let mut bytes = name.bytes();
134            if name.len() > 128
135                || !bytes
136                    .next()
137                    .is_some_and(|b| b.is_ascii_alphabetic() || b == b'_' || b == b'$')
138                || !bytes.all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'$')
139            {
140                return Err("app.utilities: invalid service name".to_owned());
141            }
142            let path = format!("app.utilities.{name}");
143            let definition = object(definition, &path)?;
144            known_fields(definition, &path, &["configuration"])?;
145            if let Some(config) = definition.get("configuration")
146                && serde_json::to_vec(config).map_err(|e| e.to_string())?.len() > 64 * 1024
147            {
148                return Err(format!("{path}.configuration: exceeds 64 KiB"));
149            }
150        }
151    }
152    nonempty_string(app.get("identifier"), "app.identifier")?;
153    let security = object(required(app, "security", "app.security")?, "app.security")?;
154    known_fields(security, "app.security", &["csp", "capabilities"])?;
155    check_origin_capabilities(required(
156        security,
157        "capabilities",
158        "app.security.capabilities",
159    )?)?;
160    if let Some(csp) = security.get("csp") {
161        let csp = nonempty_string(Some(csp), "app.security.csp")?;
162        if csp.len() > 4096 || csp.contains(['\r', '\n']) {
163            return Err("app.security.csp: expected 1..=4096 bytes without newlines".to_owned());
164        }
165    }
166    if let Some(assets) = app.get("assets") {
167        check_assets(assets, namespace)?;
168    }
169    if let Some(network) = app.get("network") {
170        check_network(network)?;
171    }
172    if let Some(windows) = app.get("windows") {
173        check_windows(windows)?;
174    }
175    Ok(())
176}

178fn check_network(value: &Value) -> Result<(), String> {

180    let network = object(value, path)?;
181    known_fields(
182        network,
183        path,
184        &["disableHttp2", "maxResponseBytes", "timeoutSeconds"],
185    )?;

187        boolean(Some(value), "app.network.disableHttp2")?;
188    }
189    for (name, maximum) in [
190        ("maxResponseBytes", 5 * 1024 * 1024),
191        ("timeoutSeconds", 300),

195            if value == 0 || value > maximum {
196                return Err(format!("{path}.{name}: expected integer in 1..={maximum}"));
197            }
198        }
199    }
200    Ok(())

203fn check_windows(value: &Value) -> Result<(), String> {
204    let windows = array(value, "app.windows")?;
205    if windows.is_empty() {
206        return Err("app.windows must not be empty".to_owned());
207    }
208    let mut labels = BTreeSet::new();
209    for (index, window) in windows.iter().enumerate() {
210        let path = format!("app.windows[{index}]");
211        let window = object(window, &path)?;
212        known_fields(
213            window,
214            &path,
215            &[
216                "label",
217                "url",
218                "title",
219                "x",
220                "y",
221                "width",
222                "height",
223                "minWidth",
224                "minHeight",
225                "maxWidth",
226                "maxHeight",
227                "focused",
228                "decorations",
229            ],
230        )?;
231        let label = nonempty_string(window.get("label"), &format!("{path}.label"))?;
232        if !labels.insert(label) {
233            return Err(format!("{path}.label: duplicate window label"));
234        }
235        nonempty_string(window.get("url"), &format!("{path}.url"))?;
236        for name in ["width", "height"] {
237            let field = format!("{path}.{name}");
238            let value = integer(window.get(name), &field)?;
239            if value == 0 || value > 16_384 {
240                return Err(format!("{field}: expected integer in 1..=16384"));
241            }
242        }
243        for (dimension, min, max) in [
244            ("width", "minWidth", "maxWidth"),
245            ("height", "minHeight", "maxHeight"),
246        ] {
247            let read = |key: &str, default| -> Result<u64, String> {
248                let value = match window.get(key) {
249                    Some(v) => integer(Some(v), &format!("{path}.{key}"))?,
250                    None => default,
251                };
252                if value == 0 || value > 16384 {
253                    return Err(format!("{path}.{key}: expected integer in 1..=16384"));
254                }
255                Ok(value)
256            };
257            let low = read(min, 1)?;
258            let high = read(max, 16384)?;
259            let size = window[dimension].as_u64().unwrap();
260            if low > high || size < low || size > high {
261                return Err(format!("{path}: inconsistent {dimension} constraints"));
262            }
263        }
264        if window.contains_key("x") != window.contains_key("y") {
265            return Err(format!("{path}: x and y must be declared together"));
266        }
267        for name in ["focused", "decorations"] {
268            if let Some(value) = window.get(name) {
269                boolean(Some(value), &format!("{path}.{name}"))?;
270            }
271        }
272    }
273    Ok(())
274}

276fn origins_overlap(left: &str, right: &str) -> bool {
277    let sample = |origin: &str| url::Url::parse(&origin.replacen("*.", "sample.", 1)).unwrap();
278    let matches = |pattern: &str, candidate: &url::Url| {
279        let base = sample(pattern);
280        if base.scheme() != candidate.scheme() || base.port() != candidate.port() {
281            return false;
282        }
283        let host = candidate.host_str().unwrap().to_ascii_lowercase();
284        let expected = base.host_str().unwrap().to_ascii_lowercase();
285        if pattern.contains("://*.") {
286            host.ends_with(&expected[6..]) && host.len() > expected.len() - 6
287        } else {
288            host == expected
289        }
290    };
291    matches(left, &sample(right)) || matches(right, &sample(left))
292}

294fn check_assets(value: &Value, scheme: &str) -> Result<(), String> {
295    let mut routes: Vec<(String, String)> = vec![];
296    for (index, route) in array(value, "app.assets")?.iter().enumerate() {
297        let path = format!("app.assets[{index}]");
298        let route = object(route, &path)?;
299        known_fields(route, &path, &["origins", "paths", "file", "directory"])?;
300        strings(
301            required(route, "origins", &path)?,
302            &format!("{path}.origins"),
303            false,
304        )?;
305        strings(
306            required(route, "paths", &path)?,
307            &format!("{path}.paths"),
308            false,
309        )?;
310        if route.contains_key("file") == route.contains_key("directory") {
311            return Err(format!("{path}: specify exactly one of file/directory"));
312        }
313        let source = nonempty_string(route.get("file").or_else(|| route.get("directory")), &path)?;
314        if source.contains(['\\', ':'])
315            || std::path::Path::new(source)
316                .components()
317                .any(|part| !matches!(part, std::path::Component::Normal(_)))
318        {
319            return Err(format!(
320                "{path}: asset source must be a safe project-relative path"
321            ));
322        }
323        for origin in route["origins"].as_array().unwrap() {
324            let origin = origin.as_str().unwrap();
325            check_origin_pattern(origin, &path, false)?;
326            if origin.split("://").next() != Some(scheme) {
327                return Err(format!("{path}: assets must use the application scheme"));
328            }
329            for pattern in route["paths"].as_array().unwrap() {
330                let pattern = pattern.as_str().unwrap();
331                if matches!(pattern, "/tuffite-runtime.js" | "/tuffite-shell-api.js") {
332                    return Err(format!("{path}: reserved framework route"));
333                }
334                let prefix = pattern.strip_suffix('*').unwrap_or(pattern);
335                if !prefix.starts_with('/')
336                    || prefix.contains(['*', '?', '#', '\\', '%'])
337                    || prefix
338                        .split('/')
339                        .any(|segment| matches!(segment, "." | ".."))
340                    || (pattern.ends_with('*') && !prefix.ends_with('/'))
341                    || (route.contains_key("directory") && !pattern.ends_with('*'))
342                {
343                    return Err(format!(
344                        "{path}: paths must be exact paths or a terminal /* prefix; directories require /*"
345                    ));
346                }
347                if routes
348                    .iter()
349                    .any(|(o, p)| p == pattern && origins_overlap(origin, o))
350                {
351                    return Err(format!("{path}: ambiguous asset route {pattern}"));
352                }
353                routes.push((origin.to_owned(), pattern.to_owned()));
354            }
355        }
356    }
357    Ok(())
358}

380fn check_origin_pattern(pattern: &str, path: &str, allow_self: bool) -> Result<(), String> {
381    if pattern == "*" || (allow_self && pattern == "self") {
382        return Ok(());
383    }
384    let fail = || format!("{path}: expected an origin with an optional leading *. hostname");
385    if pattern.chars().any(char::is_whitespace) || pattern.contains([',', '\\']) {
386        return Err(fail());
387    }
388    let (scheme, authority) = pattern.split_once("://").ok_or_else(fail)?;
389    let candidate = if let Some(host) = authority.strip_prefix("*.") {
390        format!("{scheme}://wildcard.{host}")
391    } else {
392        pattern.to_owned()
393    };
394    if candidate.contains('*') {
395        return Err(fail());
396    }
397    let url = url::Url::parse(&candidate).map_err(|_| fail())?;
398    if url.host_str().is_none()
399        || url.username() != ""
400        || url.password().is_some()
401        || url.query().is_some()
402        || url.fragment().is_some()
403        || !matches!(url.path(), "" | "/")
404        || matches!(scheme, "file" | "data" | "blob" | "about" | "javascript")
405    {
406        return Err(fail());
407    }
408    Ok(())
409}

411fn check_origin_capabilities(value: &Value) -> Result<(), String> {
412    let path = "app.security.capabilities";
413    let mut identifiers = BTreeSet::new();
414    for (index, rule) in array(value, path)?.iter().enumerate() {
415        let path = format!("{path}[{index}]");
416        let rule = object(rule, &path)?;
417        known_fields(
418            rule,
419            &path,
420            &[
421                "identifier",
422                "description",
423                "origins",
424                "shellApi",
425                "explorer",
426            ],
427        )?;
428        let identifier = nonempty_string(rule.get("identifier"), &format!("{path}.identifier"))?;
429        if !identifier
430            .bytes()
431            .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'$' || b == b'-')
432            || !identifiers.insert(identifier)
433        {
434            return Err(format!("{path}.identifier: expected unique identifier"));
435        }
436        if let Some(description) = rule.get("description") {
437            nonempty_string(Some(description), &format!("{path}.description"))?;
438        }
439        strings(
440            required(rule, "origins", &format!("{path}.origins"))?,
441            &format!("{path}.origins"),
442            false,
443        )?;
444        for origin in rule["origins"].as_array().unwrap() {
445            check_origin_pattern(origin.as_str().unwrap(), &format!("{path}.origins"), false)?;
446        }
447        if let Some(value) = rule.get("shellApi") {
448            for (i, entry) in array(value, &format!("{path}.shellApi"))?
449                .iter()
450                .enumerate()
451            {
452                let field = format!("{path}.shellApi[{i}]");
453                let method = if entry.is_string() {
454                    nonempty_string(Some(entry), &field)?
455                } else {
456                    let object = object(entry, &field)?;
457                    known_fields(object, &field, &["method", "allow", "deny"])?;
458                    for key in ["allow", "deny"] {
459                        if let Some(value) = object.get(key) {
460                            array(value, &format!("{field}.{key}"))?;
461                        }
462                    }
463                    nonempty_string(object.get("method"), &format!("{field}.method"))?
464                };
465                let exact = method.strip_suffix(".*").unwrap_or(method);
466                if method != "*"
467                    && ((!method.ends_with(".*") && !exact.contains('.'))
468                        || exact.split('.').any(|part| {
469                            part.is_empty()
470                                || !part
471                                    .bytes()
472                                    .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'$')
473                        }))
474                {
475                    return Err(format!(
476                        "{field}: expected *, qualified method or trailing .* pattern"
477                    ));
478                }
479            }
480        }
481        if let Some(value) = rule.get("explorer") {
482            let field = format!("{path}.explorer");
483            let explorer = object(value, &field)?;
484            known_fields(explorer, &field, &["create", "executeJavaScript"])?;
485            for (key, value) in explorer {
486                boolean(Some(value), &format!("{field}.{key}"))?;
487            }
488        }
489    }
490    Ok(())
491}

518fn check_bundle(value: &Value) -> Result<(), String> {
519    let bundle = object(value, "bundle")?;
520    known_fields(bundle, "bundle", &["executable", "resources", "icons"])?;
521    if let Some(icons) = bundle.get("icons") {
522        let icons = object(icons, "bundle.icons")?;
523        known_fields(icons, "bundle.icons", &["macos", "windows"])?;

526                let field = format!("bundle.icons.{platform}");

529                if path.extension().and_then(std::ffi::OsStr::to_str) != Some(extension) {
530                    return Err(format!("{field}: requires a .{extension} file"));
531                }
532            }
533        }
534    }
535    if let Some(resources) = bundle.get("resources") {
536        strings(resources, "bundle.resources", true)?;
537    }
538    Ok(())
539}

541fn known_fields(value: &Map<String, Value>, path: &str, allowed: &[&str]) -> Result<(), String> {
542    for name in value.keys() {
543        if !allowed.contains(&name.as_str()) {
544            let suggestion = allowed
545                .iter()
546                .min_by_key(|candidate| edit_distance(name, candidate));
547            let hint = suggestion
548                .filter(|candidate| edit_distance(name, candidate) <= 3)
549                .map_or(String::new(), |candidate| {
550                    format!("; did you mean {candidate:?}?")
551                });
552            return Err(format!("{path}.{name}: unknown field{hint}"));
553        }
554    }
555    Ok(())
556}

558fn edit_distance(left: &str, right: &str) -> usize {
559    let mut costs: Vec<usize> = (0..=right.len()).collect();
560    for (i, a) in left.bytes().enumerate() {
561        let mut diagonal = i;
562        costs[0] = i + 1;
563        for (j, b) in right.bytes().enumerate() {
564            let old = costs[j + 1];
565            costs[j + 1] = if a == b {
566                diagonal
567            } else {
568                1 + diagonal.min(costs[j]).min(old)
569            };
570            diagonal = old;
571        }
572    }
573    costs[right.len()]
574}

617fn strings(value: &Value, path: &str, empty_allowed: bool) -> Result<(), String> {
618    let values = array(value, path)?;

620        return Err(format!("{path}: expected non-empty array"));
621    }
622    for (index, value) in values.iter().enumerate() {

625    Ok(())

628fn one_of(value: &Value, path: &str, allowed: &[&str]) -> Result<(), String> {
629    let value = nonempty_string(Some(value), path)?;
630    if allowed.contains(&value) {
631        Ok(())
632    } else {
633        Err(format!("{path}: expected one of {}", allowed.join(", ")))
634    }
635}
