26pub(super) fn configure_assets(
27    project_dir: &Path,
28    app: &PackageApp,
29    bundle_resources: &[String],
30    _packaged_manifest: &mut serde_json::Value,
31) -> Result<Vec<String>, String> {
32    let mut resources = bundle_resources.to_vec();
33    for route in &app.assets {
34        let (source, directory) = match (&route.file, &route.directory) {
35            (Some(file), None) => (file, false),
36            (None, Some(directory)) => (directory, true),
37            _ => return Err("app.assets: specify exactly one of file/directory".to_owned()),
38        };
39        let relative = safe_relative_path(source, "app.assets source")?;
40        let path = project_dir.join(&relative);
41        if directory {
42            canonical_directory(&path)?;
43        } else {
44            canonical_file(&path)?;
45        }
46        let source = portable_relative_path(&relative)?;
47        if !resources.iter().any(|entry| entry == &source) {
48            resources.push(source);
49        }
50    }
51    // A directory already includes its children. Avoid duplicate copies.
52    let all = resources.clone();
53    resources.retain(|entry| {
54        !all.iter()
55            .any(|parent| parent != entry && entry.starts_with(&format!("{parent}/")))
56    });
57    Ok(resources)
58}

60pub(super) fn copy_resources(
61    project_dir: &Path,
62    asset_root: &Path,
63    resources: &[String],
64    package_dir: &Path,
65) -> Result<(), String> {
66    let project_dir =
67        fs::canonicalize(project_dir).map_err(io_error("resolve project directory"))?;
68    for resource in resources {
69        let relative = safe_relative_path(resource, "bundle.resources entry")?;
70        if relative == Path::new("app.icns")
71            || relative == Path::new("app.ico")
72            || relative == Path::new(RENDERER_RUNTIME_FILE)
73            || relative == Path::new(FRAMEWORK_SHELL_API_FILE)
74        {
75            return Err(format!(
76                "bundle.resources entry is reserved by Tuffite: {resource}"
77            ));
78        }
79        let unresolved = project_dir.join(&relative);
80        reject_path_symlinks(&project_dir, &relative)?;
81        let source = fs::canonicalize(&unresolved)
82            .map_err(|error| format!("resolve resource {}: {error}", unresolved.display()))?;
83        if !source.starts_with(&project_dir) {
84            return Err(format!("resource escaped project directory: {resource}"));
85        }
86        let target = package_dir.join(asset_root).join(&relative);
87        if source.is_file() {
88            copy_regular_file(&source, &target)?;
89        } else if source.is_dir() {
90            copy_directory(&source, &target)?;
91        } else {
92            return Err(format!(
93                "bundle resource is neither a regular file nor a directory: {resource}"
94            ));
95        }
96    }
97    Ok(())
98}

100pub(super) fn reject_path_symlinks(root: &Path, relative: &Path) -> Result<(), String> {
101    let mut current = root.to_path_buf();
102    for component in relative.components() {
103        let Component::Normal(part) = component else {
104            return Err(format!("unsafe relative path: {}", relative.display()));
105        };
106        current.push(part);
107        reject_symlink(&current)?;
108    }
109    Ok(())
110}

112pub(super) fn copy_regular_file(source: &Path, target: &Path) -> Result<(), String> {
113    reject_symlink(source)?;
114    if !source.is_file() {
115        return Err(format!("expected a regular file: {}", source.display()));
116    }
117    if let Some(parent) = target.parent() {
118        fs::create_dir_all(parent).map_err(|error| {
119            format!(
120                "create package subdirectory {} for {}: {error}",
121                parent.display(),
122                target.display()
123            )
124        })?;
125    }
126    fs::copy(source, target)
127        .map(|_| ())
128        .map_err(|error| format!("copy {} to {}: {error}", source.display(), target.display()))
129}

131pub(super) fn copy_directory(source: &Path, target: &Path) -> Result<(), String> {
132    let relative_files = collect_files(source)?;
133    for relative in relative_files {
134        copy_regular_file(&source.join(&relative), &target.join(relative))?;
135    }
136    Ok(())
137}

139pub(super) fn validate_internal_asset_root(
140    asset_root: &Path,
141    executable_name: &str,
142    runtime_library: &str,
143) -> Result<(), String> {
144    let root = asset_root
145        .components()
146        .next()
147        .and_then(|component| match component {
148            Component::Normal(part) => part.to_str(),
149            _ => None,
150        })
151        .ok_or_else(|| {
152            "internal asset root must start with a valid UTF-8 directory name".to_owned()
153        })?;
154    let launcher = if cfg!(target_os = "windows") {
155        "run-tuffite.cmd"
156    } else {
157        "run-tuffite.sh"
158    };
159    for reserved in [
160        executable_name,
161        runtime_library,
162        "tuffite.json",
163        MANIFEST_FILE,
164        "README.txt",
165        launcher,
166    ] {
167        if root == reserved {
168            return Err(format!(
169                "internal asset root top-level directory `{root}` conflicts with package-root file `{reserved}`"
170            ));
171        }
172    }
173    Ok(())
174}
