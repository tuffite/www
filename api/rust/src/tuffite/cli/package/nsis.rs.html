1//! CLI-owned, versioned NSIS toolchain; application users do not need Mise.
2use super::super::filesystem::{canonical_directory, io_error, reject_symlink};
3use sha2::{Digest, Sha256};
4use std::env;
5use std::fs::{self, File, OpenOptions};
6use std::io::{self, Cursor, Read};
7use std::path::{Component, Path, PathBuf};
8use std::process::Command;

10const VERSION: &str = "3.11";
11const URL: &str = "https://downloads.sourceforge.net/project/nsis/NSIS%203/3.11/nsis-3.11.zip";
12const SHA256: &str = "c7d27f780ddb6cffb4730138cd1591e841f4b7edb155856901cdf5f214394fa1";
13const ARCHIVE_LIMIT: u64 = 8 * 1024 * 1024;
14const EXTRACT_LIMIT: u64 = 64 * 1024 * 1024;
15const REQUIRED: &[&str] = &[
16    "makensis.exe",
17    "Bin/makensis.exe",
18    "Stubs/lzma_solid-x86-unicode",
19    "Include/LogicLib.nsh",
20    "Include/MUI2.nsh",
21    "Plugins/x86-unicode/System.dll",
22];

24pub(super) struct Nsis {
25    directory: PathBuf,
26}

28impl Nsis {
29    pub(super) fn resolve() -> Result<Self, String> {
30        if let Some(directory) = env::var_os("TUFFITE_NSIS_DIR") {
31            return Self::open(Path::new(&directory)).map_err(|error| {
32                format!("invalid TUFFITE_NSIS_DIR: {error}; provide the complete NSIS directory")
33            });
34        }
35        let cache = if let Some(path) = env::var_os("TUFFITE_TOOLS_CACHE") {
36            PathBuf::from(path)
37        } else if cfg!(windows) {
38            PathBuf::from(
39                env::var_os("LOCALAPPDATA")
40                    .ok_or("LOCALAPPDATA is unavailable; set TUFFITE_TOOLS_CACHE")?,
41            )
42            .join("Tuffite/tools")
43        } else {
44            return Err("automatic NSIS provisioning requires a Windows host".into());
45        };
46        let offline = match env::var("TUFFITE_TOOLS_OFFLINE") {
47            Err(env::VarError::NotPresent) => false,
48            Ok(value) if matches!(value.as_str(), "1" | "true") => true,
49            Ok(value) if matches!(value.as_str(), "0" | "false") => false,
50            _ => return Err("TUFFITE_TOOLS_OFFLINE must be true/false or 1/0".into()),
51        };
52        Self::cached(&cache, offline, Self::download)
53    }
54
55    fn open(directory: &Path) -> Result<Self, String> {
56        let directory = canonical_directory(directory)?;
57        for name in REQUIRED {
58            let path = directory.join(name);
59            reject_symlink(&path)?;
60            if !path.is_file() {
61                return Err(format!("NSIS toolchain is missing {name}"));
62            }
63        }
64        Ok(Self { directory })
65    }
66
67    pub(super) fn command(&self) -> Command {
68        let mut command = Command::new(self.directory.join("makensis.exe"));
69        // Prevent an inherited installation path from mixing other stubs/plugins.
70        command.env("NSISDIR", &self.directory);
71        command
72    }
73
74    fn download() -> Result<Vec<u8>, String> {
75        eprintln!("Downloading NSIS {VERSION} (first installer build)...");
76        let agent: ureq::Agent = ureq::Agent::config_builder()
77            .timeout_global(Some(std::time::Duration::from_secs(120)))
78            .build()
79            .into();
80        let mut response = agent.get(URL).call().map_err(|error| {
81            format!("download NSIS: {error}; use TUFFITE_NSIS_DIR for an offline installation")
82        })?;
83        let mut bytes = Vec::new();
84        response
85            .body_mut()
86            .as_reader()
87            .take(ARCHIVE_LIMIT + 1)
88            .read_to_end(&mut bytes)
89            .map_err(io_error("download NSIS archive"))?;
90        if bytes.len() as u64 > ARCHIVE_LIMIT {
91            return Err("NSIS archive exceeds the download limit".into());
92        }
93        Ok(bytes)
94    }
95
96    fn cached(
97        cache: &Path,
98        offline: bool,
99        download: impl FnOnce() -> Result<Vec<u8>, String>,
100    ) -> Result<Self, String> {
101        fs::create_dir_all(cache).map_err(io_error("create tools cache"))?;
102        let lock_path = cache.join(format!("nsis-{VERSION}.lock"));
103        if lock_path.exists() {
104            reject_symlink(&lock_path)?;
105        }
106        let lock = OpenOptions::new()
107            .create(true)
108            .truncate(false)
109            .read(true)
110            .write(true)
111            .open(lock_path)
112            .map_err(io_error("open NSIS cache lock"))?;
113        lock.lock().map_err(io_error("lock NSIS cache"))?;
114        let directory = cache.join(format!("nsis-{VERSION}"));
115        if directory.exists() {
116            reject_symlink(&directory)?;
117        }
118        if let Ok(tool) = Self::open(&directory) {
119            return Ok(tool);
120        }
121
122        let archive_path = cache.join(format!("nsis-{VERSION}.zip"));
123        let bytes = if archive_path.exists() {
124            reject_symlink(&archive_path)?;
125            let bytes = fs::read(&archive_path).map_err(io_error("read cached NSIS archive"))?;
126            if Self::verify(&bytes).is_ok() {
127                bytes
128            } else if offline {
129                return Err("cached NSIS archive failed SHA-256 verification; replace it or unset TUFFITE_TOOLS_OFFLINE".into());
130            } else {
131                download()?
132            }
133        } else if offline {
134            return Err(format!(
135                "NSIS {VERSION} is not cached; unset TUFFITE_TOOLS_OFFLINE for the first installer build, or set TUFFITE_NSIS_DIR"
136            ));
137        } else {
138            download()?
139        };
140        Self::verify(&bytes)?;
141        // The lock serializes cache repair. A interrupted archive write is rejected
142        // on the next invocation, and the directory is published only when complete.
143        fs::write(&archive_path, &bytes).map_err(io_error("cache NSIS archive"))?;
144        let staging = cache.join(format!("nsis-{VERSION}.staging"));
145        if staging.exists() {
146            reject_symlink(&staging)?;
147            fs::remove_dir_all(&staging).map_err(io_error("remove interrupted NSIS extraction"))?;
148        }
149        let result = (|| {
150            Self::extract(&bytes, &staging)?;
151            Self::open(&staging)?;
152            if directory.exists() {
153                fs::remove_dir_all(&directory).map_err(io_error("remove incomplete NSIS cache"))?;
154            }
155            fs::rename(&staging, &directory).map_err(io_error("publish NSIS toolchain"))?;
156            Self::open(&directory)
157        })();
158        if result.is_err() {
159            let _ = fs::remove_dir_all(staging);
160        }
161        result
162    }
163
164    fn verify(bytes: &[u8]) -> Result<(), String> {
165        if format!("{:x}", Sha256::digest(bytes)) != SHA256 {
166            return Err("NSIS archive failed SHA-256 verification; no tools were installed".into());
167        }
168        Ok(())
169    }
170
171    fn extract(bytes: &[u8], destination: &Path) -> Result<(), String> {
172        let mut archive = zip::ZipArchive::new(Cursor::new(bytes))
173            .map_err(|error| format!("read NSIS ZIP: {error}"))?;
174        fs::create_dir(destination).map_err(io_error("create NSIS staging"))?;
175        let mut total = 0_u64;
176        for index in 0..archive.len() {
177            let mut file = archive
178                .by_index(index)
179                .map_err(|error| format!("read NSIS ZIP entry: {error}"))?;
180            let name = file.name();
181            if name.contains('\\') || name.contains(':') {
182                return Err("unsafe NSIS archive path".into());
183            }
184            let path = file.enclosed_name().ok_or("unsafe NSIS archive path")?;
185            let mut components = path.components();
186            if components.next() != Some(Component::Normal(std::ffi::OsStr::new("nsis-3.11"))) {
187                return Err("unexpected NSIS archive root".into());
188            }
189            let relative = components.as_path();
190            if relative
191                .components()
192                .any(|component| !matches!(component, Component::Normal(_)))
193                || file
194                    .unix_mode()
195                    .is_some_and(|mode| mode & 0o170_000 == 0o120_000)
196            {
197                return Err("unsafe NSIS archive entry".into());
198            }
199            let output = destination.join(relative);
200            if file.is_dir() {
201                fs::create_dir_all(output).map_err(io_error("create NSIS directory"))?;
202                continue;
203            }
204            total = total
205                .checked_add(file.size())
206                .filter(|total| *total <= EXTRACT_LIMIT)
207                .ok_or("NSIS archive exceeds extraction limit")?;
208            fs::create_dir_all(output.parent().ok_or("NSIS entry has no parent")?)
209                .map_err(io_error("create NSIS directory"))?;
210            let mut writer = File::create(output).map_err(io_error("extract NSIS file"))?;
211            let written = io::copy(&mut (&mut file).take(EXTRACT_LIMIT + 1), &mut writer)
212                .map_err(io_error("extract NSIS file"))?;
213            if written != file.size() {
214                return Err("invalid NSIS archive entry size".into());
215            }
216        }
217        Ok(())
218    }

226    fn archive(name: &str) -> Vec<u8> {
227        let mut zip = zip::ZipWriter::new(Cursor::new(Vec::new()));
228        zip.start_file(name, zip::write::SimpleFileOptions::default())
229            .unwrap();
230        zip.write_all(b"fixture").unwrap();
231        zip.finish().unwrap().into_inner()
232    }

234    #[test]
235    fn archives_reject_wrong_hash_and_unsafe_paths() {
236        let temporary = env::temp_dir().join(format!("tuffite-nsis-paths-{}", std::process::id()));
237        for name in [
238            "nsis-3.11/../../outside",
239            "nsis-3.11/C:/outside",
240            "nsis-3.11\\outside",
241            "other/makensis.exe",
242        ] {
243            let _ = fs::remove_dir_all(&temporary);
244            assert!(Nsis::extract(&archive(name), &temporary).is_err());
245        }
246        let _ = fs::remove_dir_all(temporary);
247        assert!(Nsis::verify(b"unexpected archive").is_err());
248    }

250    #[test]
251    #[ignore = "requires TUFFITE_TEST_NSIS_ARCHIVE containing the pinned official ZIP"]
252    fn verified_archive_installs_and_repairs_cache_offline() {
253        let bytes =
254            fs::read(env::var_os("TUFFITE_TEST_NSIS_ARCHIVE").expect("archive fixture")).unwrap();
255        Nsis::verify(&bytes).unwrap();
256        let cache = env::temp_dir().join(format!("tuffite-nsis-real-{}", std::process::id()));
257        let _ = fs::remove_dir_all(&cache);
258        let first = Nsis::cached(&cache, false, || Ok(bytes)).unwrap();
259        assert!(
260            first
261                .directory
262                .join("makensis.exe")
263                .metadata()
264                .unwrap()
265                .len()
266                > 0
267        );
268        fs::remove_file(first.directory.join(REQUIRED[2])).unwrap();
269        let repaired = Nsis::cached(&cache, true, || {
270            panic!("repair must reuse the verified ZIP")
271        })
272        .unwrap();
273        assert!(repaired.directory.join(REQUIRED[2]).is_file());
274        fs::remove_dir_all(cache).unwrap();
275    }

277    #[test]
278    fn complete_cache_is_reused_offline_and_partial_cache_fails_without_download() {
279        let cache = env::temp_dir().join(format!("tuffite-nsis-cache-{}", std::process::id()));
280        let _ = fs::remove_dir_all(&cache);
281        let directory = cache.join(format!("nsis-{VERSION}"));
282        for name in REQUIRED {
283            let path = directory.join(name);
284            fs::create_dir_all(path.parent().unwrap()).unwrap();
285            fs::write(path, "fixture").unwrap();
286        }
287        let tool = Nsis::cached(&cache, true, || panic!("cache reuse must not download")).unwrap();
288        assert_eq!(tool.directory, fs::canonicalize(&directory).unwrap());
289        fs::remove_file(directory.join(REQUIRED[0])).unwrap();
290        let error = Nsis::cached(&cache, true, || panic!("offline must not download"))
291            .err()
292            .unwrap();
293        assert!(error.contains("not cached"));
294        fs::remove_dir_all(cache).unwrap();
295    }
