48#[derive(clap::Args, Debug)]
49pub struct Options {
50    /// Application configuration; otherwise search the current directory and its parents.
51    #[arg(long)]
52    manifest: Option<PathBuf>,
53    #[arg(long)]
54    destination: Option<PathBuf>,
55    #[arg(long)]
56    name: Option<String>,
57    #[arg(long)]
58    include_symbols: bool,
59    #[arg(long, value_enum, default_value_t = Format::Zip)]
60    format: Format,
61    /// Keep debug/local symbols in staged binaries.
62    #[arg(long)]
63    no_strip: bool,
64    /// Export private, matching application symbols separately.
65    #[arg(long)]
66    symbols_dir: Option<PathBuf>,
67    #[arg(long)]
68    force: bool,
69}

95pub(super) fn prepare_development(
96    context: &mut super::Context,
97    manifest: &Path,
98    package_dir: &Path,
99    release: bool,
100) -> Result<(), String> {
101    let destination = package_dir
102        .parent()
103        .ok_or_else(|| "development package has no parent directory".to_owned())?;
104    let name = package_dir
105        .file_name()
106        .and_then(OsStr::to_str)
107        .ok_or_else(|| "development package name must be valid UTF-8".to_owned())?;
108    Packager::new(
109        context,
110        &Options {
111            manifest: Some(manifest.to_path_buf()),
112            destination: Some(destination.to_path_buf()),
113            name: Some(name.to_owned()),
114            include_symbols: false,
115            format: Format::Directory,
116            no_strip: true,
117            symbols_dir: None,
118            force: true,
119        },
120        Mode::Development { release },
121    )
122    .run()
123}

125#[derive(Clone, Copy)]
126enum Mode {
127    Distribution,
128    Development { release: bool },
129}

131struct Packager<'a> {
132    context: &'a mut super::Context,
133    options: &'a Options,
134    mode: Mode,
135}
136
137impl<'a> Packager<'a> {
138    fn new(context: &'a mut super::Context, options: &'a Options, mode: Mode) -> Self {
139        Self {
140            context,
141            options,
142            mode,
143        }
144    }
145
146    #[allow(clippy::too_many_lines)]
147    fn run(self) -> Result<(), String> {
148        let Self {
149            context,
150            options,
151            mode,
152        } = self;
153        let (create_archive, release) = match mode {
154            Mode::Distribution => (options.format.archive_extension().is_some(), true),
155            Mode::Development { release } => (false, release),
156        };
157        options.format.validate()?;
158        let manifest_path = super::config::resolve_manifest(options.manifest.as_deref())?;
159        tracing::info!(
160            manifest = %manifest_path.display(),
161            "packaging Tuffite application"
162        );
163        super::config::check(&manifest_path)?;
164
165        let project_dir = manifest_path
166            .parent()
167            .ok_or_else(|| "manifest has no parent directory".to_owned())?;
168        let mut packaged_manifest: serde_json::Value = serde_json::from_reader(BufReader::new(
169            File::open(&manifest_path).map_err(io_error("open application manifest"))?,
170        ))
171        .map_err(|error| format!("parse {}: {error}", manifest_path.display()))?;
172        let command_ids = crate::build::security_command_ids(&manifest_path)?;
173        if let Some(rules) = packaged_manifest["app"]["security"]["capabilities"].as_array_mut() {
174            for rule in rules {
175                if let Some(entries) = rule
176                    .get_mut("shellApi")
177                    .and_then(serde_json::Value::as_array_mut)
178                {
179                    let mut compiled = Vec::new();
180                    for entry in entries.iter() {
181                        compiled.extend(crate::build::compile_security_entry(entry, &command_ids)?);
182                    }
183                    *entries = compiled;
184                }
185            }
186        }
187        // Runtime metadata contains identities, never build-time declaration paths.
188        packaged_manifest["app"]["shellApi"] = serde_json::json!({
189            "namespace": packaged_manifest["build"]["shellApi"]["namespace"],
190            "commandIds": command_ids,
191        });
192        packaged_manifest
193            .as_object_mut()
194            .expect("validated manifest")
195            .remove("build");
196        let app_manifest: AppManifest = serde_json::from_value(packaged_manifest.clone())
197            .map_err(|error| format!("parse {}: {error}", manifest_path.display()))?;
198        validate_application_id(&app_manifest.app.identifier)?;
199        let framework = context.framework()?;
200        let framework_manifest = framework.manifest();
201        tracing::info!(
202            framework = %framework.root().display(),
203            framework_version = %framework_manifest.framework_version,
204            ffi_abi = %format_args!("{}.{}.{}", framework_manifest.ffi_abi_major, framework_manifest.ffi_abi_minor, framework_manifest.ffi_abi_patch),
205            target = %framework_manifest.target,
206            profile = %framework_manifest.profile,
207            variant = %framework_manifest.variant,
208            development_enabled = framework_manifest.development_enabled,
209            "using Tuffite Framework Distribution"
210        );
211        let target = package_target(&framework.manifest().target)?;
212        let application_runtime = ApplicationRuntime::build(project_dir, framework, release)?;
213        let product_name = app_manifest
214            .product_name
215            .as_deref()
216            .unwrap_or(&application_runtime.package_name);
217        super::config::validate_product_name(product_name)?;
218        packaged_manifest["productName"] = serde_json::Value::String(product_name.to_owned());
219        tracing::info!(
220            path = %application_runtime.library.display(),
221            target = target.triple,
222            profile = if release { "release" } else { "debug" },
223            "built application runtime"
224        );
225        let resources = configure_assets(
226            project_dir,
227            &app_manifest.app,
228            &app_manifest.bundle.resources,
229            &mut packaged_manifest,
230        )
231        .map_err(|error| {
232            format!("{error}; build the application frontend after Cargo has run build.rs, then retry packaging")
233        })?;
234        reject_symlink(&application_runtime.library)?;
235        if let Some(configured) = app_manifest.bundle.executable.as_deref() {
236            validate_executable_stem(configured)?;
237        }
238        let executable_name = executable_name(
239            app_manifest.bundle.executable.as_deref(),
240            &application_runtime.package_name,
241        );
242        validate_executable_name(&executable_name)?;
243        let runtime_library = application_runtime
244            .library
245            .file_name()
246            .and_then(OsStr::to_str)
247            .ok_or_else(|| "Cargo cdylib filename must be valid UTF-8".to_owned())?
248            .to_owned();
249        validate_runtime_library_name(&runtime_library)?;
250        let asset_root = PathBuf::from(ASSET_ROOT);
251        validate_internal_asset_root(&asset_root, &executable_name, &runtime_library)?;
252
253        let package_name = options.name.clone().unwrap_or_else(|| {
254            if options.format.is_bundle() {
255                product_name.to_owned()
256            } else if app_manifest.product_name.is_some() {
257                format!("{product_name}-{}-{}", target.platform, target.architecture)
258            } else {
259                default_package_name(&app_manifest.app.identifier, &target)
260            }
261        });
262        validate_package_name(&package_name)?;
263        let destination = absolute_path(
264            options
265                .destination
266                .clone()
267                .unwrap_or_else(|| project_dir.join("dist")),
268        )?;
269        fs::create_dir_all(&destination).map_err(io_error("create package destination"))?;
270        let destination = canonical_directory(&destination)?;
271        let directory_name = if options.format.is_bundle() {
272            format!("{package_name}.app")
273        } else {
274            package_name.clone()
275        };
276        let final_package_dir = destination.join(&directory_name);
277        if let Some(symbols) = &options.symbols_dir {
278            let symbols = absolute_path(symbols.clone())?;
279            let parent = symbols.parent().ok_or("symbols directory has no parent")?;
280            fs::create_dir_all(parent).map_err(io_error("create symbols parent"))?;
281            let symbols = canonical_directory(parent)?
282                .join(symbols.file_name().ok_or("invalid symbols directory")?);
283            if final_package_dir.starts_with(&symbols) || symbols.starts_with(&final_package_dir) {
284                return Err("symbols directory overlaps the final application package".into());
285            }
286        }
287        let final_archive_path = destination.join(format!(
288            "{package_name}.{}",
289            options.format.archive_extension().unwrap_or("zip")
290        ));
291        let staging_suffix = format!("{}.{}", package_name, std::process::id());
292        let staging_package_dir = destination.join(format!(
293            ".{staging_suffix}.staging{}",
294            if options.format.is_bundle() {
295                ".app"
296            } else {
297                ""
298            }
299        ));
300        let staging_archive_path = destination.join(format!(
301            ".{staging_suffix}.staging.{}",
302            options.format.archive_extension().unwrap_or("zip")
303        ));
304        let (package_dir, archive_path) = if create_archive {
305            validate_output(
306                &final_package_dir,
307                Some(final_archive_path.as_path()),
308                options.force,
309            )?;
310            if staging_package_dir.exists() {
311                reject_symlink(&staging_package_dir)?;
312                remove_dir_all_with_retry(&staging_package_dir)?;
313            }
314            if staging_archive_path.exists() {
315                reject_symlink(&staging_archive_path)?;
316                fs::remove_file(&staging_archive_path)
317                    .map_err(io_error("remove stale staging archive"))?;
318            }
319            fs::create_dir(&staging_package_dir)
320                .map_err(io_error("create staging package directory"))?;
321            (staging_package_dir.clone(), staging_archive_path.clone())
322        } else {
323            if matches!(mode, Mode::Distribution) {
324                validate_output(&final_package_dir, None, options.force)?;
325            }
326            if staging_package_dir.exists() {
327                reject_symlink(&staging_package_dir)?;
328                remove_dir_all_with_retry(&staging_package_dir)?;
329            }
330            fs::create_dir(&staging_package_dir)
331                .map_err(io_error("create development staging package directory"))?;
332            (staging_package_dir.clone(), final_archive_path.clone())
333        };
334
335        let result = (|| {
336            let runtime = copy_framework_runtime(
337                framework,
338                &package_dir,
339                &asset_root,
340                OsStr::new(&executable_name),
341                options.include_symbols,
342            )?;
343            copy_regular_file(
344                &application_runtime.library,
345                &package_dir.join(&runtime_library),
346            )?;
347            let private_symbols = if matches!(mode, Mode::Distribution) {
348                if target.platform == "macos" {
349                    formats::relocate_libraries(&package_dir, options.format.is_bundle())?;
350                }
351                symbols::SymbolProcessor {
352                    root: &package_dir,
353                    application_library: &application_runtime.library,
354                    strip: !options.no_strip,
355                    destination: options.symbols_dir.as_deref(),
356                    force: options.force,
357                    identifier: &app_manifest.app.identifier,
358                    target: &target,
359                }
360                .prepare()?
361            } else {
362                None
363            };
364            app_manifest.bundle.icons.install(
365                &package_dir,
366                project_dir,
367                target.platform,
368                &executable_name,
369            )?;
370            write_packaged_manifest(&package_dir.join("tuffite.json"), &packaged_manifest)?;
371            copy_resources(project_dir, &asset_root, &resources, &package_dir)?;
372            if !options.format.is_bundle() {
373                write_launcher(&package_dir, OsStr::new(&executable_name))?;
374            }
375            write_readme(
376                &package_dir,
377                OsStr::new(&executable_name),
378                options.format.is_bundle(),
379            )?;
380
381            write_package_manifest(
382                &package_dir,
383                &app_manifest.app.identifier,
384                executable_name.clone(),
385                runtime_library,
386                runtime.development_enabled,
387                &target,
388                framework.manifest(),
389                options.format.is_bundle(),
390            )?;
391            let bundle = formats::ApplicationBundle {
392                identifier: &app_manifest.app.identifier,
393                product_name,
394                executable: &executable_name,
395                version: &application_runtime.version,
396            };
397            if options.format.is_bundle() {
398                bundle.assemble(&package_dir)?;
399            }
400            if create_archive {
401                if options.format == Format::Zip {
402                    write_zip(&package_dir, &archive_path, &package_name)?;
403                } else {
404                    bundle.installer(
405                        options.format,
406                        &package_dir,
407                        &archive_path,
408                        &directory_name,
409                    )?;
410                }
411                publish_release(
412                    &package_dir,
413                    &archive_path,
414                    &final_package_dir,
415                    &final_archive_path,
416                )?;
417                if let Some(symbols) = private_symbols {
418                    symbols.publish()?;
419                }
420                finish_package(&final_package_dir, Some(&final_archive_path))?;
421            } else {
422                publish_development(&package_dir, &final_package_dir)?;
423                if let Some(symbols) = private_symbols {
424                    symbols.publish()?;
425                }
426                finish_package(&final_package_dir, None)?;
427            }
428            Ok(())
429        })();
430
431        if result.is_err() && package_dir == staging_package_dir {
432            let _ = fs::remove_dir_all(&package_dir);
433            if create_archive {
434                let _ = fs::remove_file(&archive_path);
435            }
436        }
437        result
438    }
439}
