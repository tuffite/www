1//! Shared filesystem validation and portable paths for CLI workflows.
2use std::env;
3use std::fs;
4use std::io;
5use std::path::{Component, Path, PathBuf};
6
7pub(super) fn absolute_path(path: PathBuf) -> Result<PathBuf, String> {
8    if path.is_absolute() {
9        Ok(path)
10    } else {
11        env::current_dir()
12            .map(|current| current.join(path))
13            .map_err(io_error("read current directory"))
14    }
15}

17pub(super) fn canonical_file(path: &Path) -> Result<PathBuf, String> {
18    reject_symlink(path)?;
19    let canonical = fs::canonicalize(path)
20        .map_err(|error| format!("resolve file {}: {error}", path.display()))?;
21    if !canonical.is_file() {
22        return Err(format!("expected a file: {}", canonical.display()));
23    }
24    Ok(canonical)
25}
26
27pub(super) fn canonical_directory(path: &Path) -> Result<PathBuf, String> {
28    let canonical = fs::canonicalize(path)
29        .map_err(|error| format!("resolve directory {}: {error}", path.display()))?;
30    if !canonical.is_dir() {
31        return Err(format!("expected a directory: {}", canonical.display()));
32    }
33    Ok(canonical)
34}

36pub(super) fn safe_relative_path(value: &str, field: &str) -> Result<PathBuf, String> {
37    if value.is_empty() {
38        return Err(format!("{field} must not be empty"));
39    }
40    let path = Path::new(value);
41    if path.is_absolute() {
42        return Err(format!("{field} must be relative: {value}"));
43    }
44    let mut normalized = PathBuf::new();
45    for component in path.components() {
46        match component {
47            Component::Normal(part) => normalized.push(part),
48            _ => return Err(format!("{field} contains an unsafe path: {value}")),
49        }
50    }
51    if normalized.as_os_str().is_empty() {
52        return Err(format!("{field} must not be empty"));
53    }
54    Ok(normalized)
55}

57pub(super) fn portable_relative_path(path: &Path) -> Result<String, String> {
58    path.components()
59        .map(|component| match component {
60            Component::Normal(part) => part
61                .to_str()
62                .map(ToOwned::to_owned)
63                .ok_or_else(|| "asset paths must be valid UTF-8".to_owned()),
64            _ => Err(format!("unsafe asset path: {}", path.display())),
65        })
66        .collect::<Result<Vec<_>, _>>()
67        .map(|parts| parts.join("/"))
68}
69
70pub(super) fn reject_symlink(path: &Path) -> Result<(), String> {
71    let metadata = fs::symlink_metadata(path)
72        .map_err(|error| format!("inspect {}: {error}", path.display()))?;
73    if metadata.file_type().is_symlink() {
74        return Err(format!(
75            "symbolic links are not allowed: {}",
76            path.display()
77        ));
78    }
79    Ok(())
80}
81
82pub(super) fn validate_relative_components(path: &Path, field: &str) -> Result<(), String> {
83    if path.as_os_str().is_empty()
84        || path
85            .components()
86            .any(|component| !matches!(component, Component::Normal(_)))
87    {
88        return Err(format!(
89            "{field} is not a safe relative path: {}",
90            path.display()
91        ));
92    }
93    Ok(())
94}

96pub(super) fn portable_path(path: &Path) -> Result<String, String> {
97    validate_relative_components(path, "package path")?;
98    path.components()
99        .map(|component| match component {
100            Component::Normal(value) => value
101                .to_str()
102                .map(ToOwned::to_owned)
103                .ok_or_else(|| format!("package path is not valid UTF-8: {}", path.display())),
104            _ => Err(format!("unsafe package path: {}", path.display())),
105        })
106        .collect::<Result<Vec<_>, _>>()
107        .map(|parts| parts.join("/"))
108}

110pub(super) fn io_error(context: &'static str) -> impl FnOnce(io::Error) -> String {
111    move |error| format!("{context}: {error}")
