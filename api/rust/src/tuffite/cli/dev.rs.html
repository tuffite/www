8use super::Context;
9use std::ffi::OsString;
10use std::path::PathBuf;
11
12#[derive(clap::Args, Debug)]
13#[allow(clippy::struct_excessive_bools)]
14pub struct Options {
15    /// Source application manifest containing the development server settings.
16    #[arg(long)]
17    manifest: Option<PathBuf>,
18    /// Unpacked development package to create or refresh.
19    #[arg(long)]
20    package: Option<PathBuf>,
21    /// Loopback HTTP Vite or other development-server entry URL.
22    #[arg(long)]
23    url: Option<String>,
24    /// Web development server executable.
25    #[arg(long)]
26    web_command: Option<PathBuf>,
27    /// Argument forwarded to the Web development server; repeat as needed.
28    #[arg(long, allow_hyphen_values = true)]
29    web_arg: Vec<OsString>,
30    /// Working directory for the Web development server.
31    #[arg(long)]
32    web_cwd: Option<PathBuf>,
33    /// Argument forwarded to the application executable; repeat as needed.
34    #[arg(long, allow_hyphen_values = true)]
35    app_arg: Vec<OsString>,
36    /// Loopback Chrome `DevTools` Protocol port. Overrides `build.devtoolsPort`.
37    #[arg(long)]
38    devtools_port: Option<u16>,
39    /// Refresh and validate the native development package without launching it.
40    #[arg(long)]
41    prepare_only: bool,
42    /// Compile the application Rust library with Cargo's release profile.
43    #[arg(long)]
44    release: bool,
45    /// Ignore local and cached Framework selections and download the Release artifact again.
46    #[arg(long)]
47    pub(super) force_download_framework: bool,
48    /// Disable automatic Rust and Shell API rebuilds when project sources change.
49    #[arg(long)]
50    no_watch: bool,
51}
52
53impl Options {
54    pub(super) fn run(&self, context: &mut Context) -> Result<(), String> {
55        session::DevelopmentSession::new(context, self)?.run()
56    }
57}
